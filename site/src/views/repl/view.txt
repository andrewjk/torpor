function getSequence(arr) {
	const p = arr.slice();
	const result = [0];

	let j;
	let u;
	let v;
	let c;

		if (arrI !== 0) {
			j = result[result.length - 1];
			if (arr[j] < arrI) {
				p[i] = j;
				result.push(i);
				continue;
			}
			u = 0;
			v = result.length - 1;
			while (u < v) {
				c = u + v >> 1;
				if (arr[result[c]] < arrI) u = c + 1;

			}
			if (arrI < arr[result[u]]) {

			}
		}
	}

		v = p[v];
	}

}
